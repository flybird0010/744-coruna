这里强调一下 生产环境部署 这里要配置跨域
/www/wwwroot/coruna1/server1/admin/.env 中
 风险：SECRET_KEY
生产环境中务必修改这个值（随便敲一段复杂的乱码即可），否则容易被伪造 JWT 令牌导致后台被黑。
CORS_ORIGINS=https://aa123.abrdns.com,http://localhost:7000,...( 增加公网域名，保留原有的)
DARKSWORD_PUBLIC_BASE  这个是攻击域名配置 例如： DARKSWORD_PUBLIC_BASE=https://aa123.abrdns.comt（公网域名）

登录认证失败 用下面的命令来重置  这里的路径一定要改
处理管理后台登录报错401
cat > /tmp/fix_login.py << 'EOF'
import sys
sys.path.insert(0, '/www/wwwroot/coruna/server')
from admin.database import SessionLocal, User, Settings
from admin.auth import get_password_hash

db = SessionLocal()
u = db.query(User).filter(User.username == 'admin').first()
if not u:
    u = User(username='admin', password=get_password_hash('admin_123'), role='admin')
    db.add(u)
    print('created new admin')
else:
    u.password = get_password_hash('admin_123')
    u.google_2fa_enabled = 0
    u.google_2fa_secret = None
    u.failed_2fa_attempts = 0
    u.locked_until = None
    u.token_version = (u.token_version or 0) + 1

for k in ['security.require_2fa', 'security.force_2fa_users']:
    s = db.query(Settings).filter(Settings.key == k).first()
    if s:
        s.value = 'false'
db.commit()
db.close()
print('OK: admin 密码已重置为 admin_123，2FA 已关闭')
EOF
